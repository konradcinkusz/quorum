    COALESCE(@SqlStatement, N'') + N'DROP TABLE [FseClub].' + QUOTENAME(TABLE_NAME) + N';' + CHAR(13)
FROM INFORMATION_SCHEMA.TABLES
WHERE TABLE_SCHEMA = 'FseClub' and TABLE_TYPE = 'BASE TABLE'
